cmake_minimum_required(VERSION 3.22)
project(myrec C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(myrec SHARED dsp.c stretch.c resample.c working_color.c lut.c dolly.c image.c focus.c asr.c stabilize.c quantize.c color.c cinema.c primary.c film_color.c halation.c noir.c sixteen_mm.c fft.c inference.c neural.c lava.c studio.c person.c)
target_sources(myrec PRIVATE p010.c background.c night.c time_film.c transition.c)
target_compile_options(myrec PRIVATE -Wall -Wextra -Werror -O3)
target_link_libraries(myrec android nativewindow jnigraphics log m dl)

set(RUNTIME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.build/runtime-sources" CACHE PATH "External runtime repositories")
set(ORT "${RUNTIME_ROOT}/onnxruntime-current")
if(NOT EXISTS "${ORT}/libonnxruntime.so")
    message(FATAL_ERROR "Run python3 scripts/prepare-runtimes.py before building")
endif()
add_library(onnxruntime SHARED IMPORTED)
set_target_properties(onnxruntime PROPERTIES IMPORTED_LOCATION "${ORT}/libonnxruntime.so")
target_include_directories(myrec PRIVATE "${ORT}")
target_link_libraries(myrec onnxruntime)
if(EXISTS "${RUNTIME_ROOT}/rnnoise-0.1/src/denoise.c")
    set(RN "${RUNTIME_ROOT}/rnnoise-0.1")
    add_library(rnnoise SHARED "${RN}/src/denoise.c" "${RN}/src/rnn.c" "${RN}/src/rnn_data.c"
        "${RN}/src/pitch.c" "${RN}/src/kiss_fft.c" "${RN}/src/celt_lpc.c")
    target_include_directories(rnnoise PRIVATE "${RN}/include" "${RN}/src")
    target_compile_definitions(rnnoise PRIVATE RNNOISE_BUILD)
    target_compile_options(rnnoise PRIVATE -O3)
    target_link_libraries(rnnoise m)
endif()
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../scripts/qwen" qwen)
