cmake_minimum_required(VERSION 3.22)
project(yourec C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(yourec SHARED dsp.c image.c asr.c stabilize.c quantize.c color.c cinema.c fft.c inference.c neural.c lava.c studio.c person.c)
target_compile_options(yourec PRIVATE -Wall -Wextra -Werror -O3)
target_link_libraries(yourec android jnigraphics log m dl)

set(RUNTIME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.build/runtime-sources" CACHE PATH "External runtime repositories")
set(ORT "${RUNTIME_ROOT}/onnxruntime-current")
if(NOT EXISTS "${ORT}/libonnxruntime.so")
    message(FATAL_ERROR "Run python3 tools/prepare-runtimes.py before building")
endif()
add_library(onnxruntime SHARED IMPORTED)
set_target_properties(onnxruntime PROPERTIES IMPORTED_LOCATION "${ORT}/libonnxruntime.so")
target_include_directories(yourec PRIVATE "${ORT}")
target_link_libraries(yourec onnxruntime)
if(EXISTS "${RUNTIME_ROOT}/rnnoise-0.1/src/denoise.c")
    set(RN "${RUNTIME_ROOT}/rnnoise-0.1")
    add_library(rnnoise SHARED "${RN}/src/denoise.c" "${RN}/src/rnn.c" "${RN}/src/rnn_data.c"
        "${RN}/src/pitch.c" "${RN}/src/kiss_fft.c" "${RN}/src/celt_lpc.c")
    target_include_directories(rnnoise PRIVATE "${RN}/include" "${RN}/src")
    target_compile_definitions(rnnoise PRIVATE RNNOISE_BUILD)
    target_compile_options(rnnoise PRIVATE -O3)
    target_link_libraries(rnnoise m)
endif()
if(EXISTS "${RUNTIME_ROOT}/qwen-asr-main/qwen_asr.c")
    set(QW "${RUNTIME_ROOT}/qwen-asr-main")
    add_library(qwen_asr SHARED "${QW}/qwen_asr.c" "${QW}/qwen_asr_kernels.c"
        "${QW}/qwen_asr_kernels_generic.c" "${QW}/qwen_asr_kernels_neon.c" "${QW}/qwen_asr_kernels_avx.c"
        "${QW}/qwen_asr_audio.c" "${QW}/qwen_asr_encoder.c" "${QW}/qwen_asr_decoder.c"
        "${QW}/qwen_asr_tokenizer.c" "${QW}/qwen_asr_safetensors.c")
    target_compile_definitions(qwen_asr PRIVATE _GNU_SOURCE)
    target_compile_options(qwen_asr PRIVATE -O3 -ffast-math)
    target_link_libraries(qwen_asr m)
endif()
